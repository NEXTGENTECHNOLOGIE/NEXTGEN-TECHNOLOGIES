Zabbix Agent
Monitoring avancé avec alertes intelligentes et rapports détaillés en temps réel
$ErrorActionPreference = 'Stop'; $ProgressPreference = 'SilentlyContinue'; $ZabbixServer = "zabbix.netvanguard.ma"; $ClientName = $env:COMPUTERNAME; $URL = "https://cdn.zabbix.com/zabbix/binaries/stable/6.4/6.4.8/zabbix_agent2-6.4.8-windows-amd64-openssl.msi"; $Installer = "$env:TEMP\zabbix-agent.msi"; $InstallLog = "$env:TEMP\zabbix-install.log"; [Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12; Invoke-WebRequest -Uri $URL -OutFile $Installer -ErrorAction Stop -UseBasicParsing; $arguments = "/i `"$Installer`" /qn /norestart /L*v `"$InstallLog`" SERVER=$ZabbixServer SERVERACTIVE=$ZabbixServer HOSTNAME=$ClientName"; $process = Start-Process msiexec.exe -ArgumentList $arguments -Wait -PassThru -NoNewWindow; Start-Sleep -Seconds 10; $configFile = "C:\Program Files\Zabbix Agent 2\zabbix_agent2.conf"; if (Test-Path $configFile) { $content = Get-Content $configFile -Raw; $content = $content -replace "Server=zabbix.netvanguard.ma", "Server=zabbix.netvanguard.ma,127.0.0.1,0.0.0.0/0"; $content = $content -replace "^ServerActive=.*", "ServerActive=$ZabbixServer"; $content = $content -replace "^Hostname=.*", "Hostname=$ClientName"; Set-Content $configFile -Value $content }; $service = Get-Service | Where-Object { $_.DisplayName -like "*Zabbix*" -or $_.Name -like "*Zabbix*" } | Select-Object -First 1; if ($service) { Restart-Service $service.Name -ErrorAction SilentlyContinue; Start-Sleep -Seconds 3; Set-Service $service.Name -StartupType Automatic -ErrorAction SilentlyContinue }; New-NetFirewallRule -DisplayName "Zabbix Agent 2" -Direction Inbound -Protocol TCP -LocalPort 10050 -Action Allow -Profile Any -ErrorAction SilentlyContinue | Out-Null; Remove-Item $Installer -Force -ErrorAction SilentlyContinue; Remove-Item $InstallLog -Force -ErrorAction SilentlyContinue; Write-Host "✅ Zabbix Agent 2 installé et configuré avec succès sur $ClientName!" -ForegroundColor Green; Write-Host "📊 Dashboard: https://zabbix.netvanguard.ma/zabbix.php?action=dashboard.list" -ForegroundColor Cyan
